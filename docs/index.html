01 — ربات تکرار (Echo)مبتدی
هر چی بگی، با ریپلای تکرار می‌کنه. پایه‌ی هر ربات.
from gitbin import Bot

TOKEN = "YOUR_BOT_TOKEN"
bot = Bot(token=TOKEN)

def main():
    bot.delete_webhook()
    offset = 0
    while True:
        updates = bot.get_updates(offset=offset, timeout=25)
        for upd in updates or []:
            offset = upd["update_id"] + 1
            if "message" not in upd:
                continue
            msg = upd["message"]
            chat_id = msg["chat"]["id"]
            message_id = msg.get("message_id")
            text = msg.get("text", "")
            if not text:
                continue
            if text == "/start":
                bot.send_message(chat_id, "سلام! هر چی بگی با ریپلای تکرار می‌کنم.",
                                 reply_to_message_id=message_id)
            else:
                bot.send_message(chat_id, f"🔁 {text}",
                                 reply_to_message_id=message_id)

if __name__ == "__main__":
    main()
02 — ربات سخنگومتوسط
یاد می‌گیره (فقط مالک) و جواب می‌ده. با یادبگیر عبارت & پاسخ.
import json, os
from gitbin import Bot

TOKEN = "YOUR_BOT_TOKEN"
OWNER_ID = 123456789
DATA_FILE = "replies.json"

bot = Bot(token=TOKEN)

def load_replies():
    if os.path.exists(DATA_FILE):
        with open(DATA_FILE, encoding="utf-8") as f:
            return json.load(f)
    return {}

def save_replies(r):
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(r, f, ensure_ascii=False, indent=2)

replies = load_replies()

def main():
    bot.delete_webhook()
    offset = 0
    while True:
        updates = bot.get_updates(offset=offset, timeout=25)
        for upd in updates or []:
            offset = upd["update_id"] + 1
            if "message" not in upd:
                continue
            msg = upd["message"]
            chat_id = msg["chat"]["id"]
            message_id = msg.get("message_id")
            text = msg.get("text", "").strip()
            is_owner = (msg.get("from", {}).get("id") == OWNER_ID)

            if text.startswith("یادبگیر") and is_owner:
                parts = text[len("یادبگیر"):].strip().split("&", 1)
                if len(parts) == 2 and parts[0].strip() and parts[1].strip():
                    replies[parts[0].strip()] = parts[1].strip()
                    save_replies(replies)
                    bot.send_message(chat_id, "✅ یاد گرفتم!",
                                     reply_to_message_id=message_id)
            elif text.startswith("فراموش") and is_owner:
                key = text[len("فراموش"):].strip()
                if key in replies:
                    del replies[key]
                    save_replies(replies)
                    bot.send_message(chat_id, "🗑️ حذف شد.",
                                     reply_to_message_id=message_id)
            elif text in replies:
                bot.send_message(chat_id, replies[text],
                                 reply_to_message_id=message_id)

if __name__ == "__main__":
    main()
03 — ربات دکمه شیشه‌ایمتوسط
منوی تعاملی با callback_query.
from gitbin import Bot
import datetime

TOKEN = "YOUR_BOT_TOKEN"
bot = Bot(token=TOKEN)

def main():
    bot.delete_webhook()
    offset = 0
    while True:
        updates = bot.get_updates(offset=offset, timeout=25)
        for upd in updates or []:
            offset = upd["update_id"] + 1
            if "message" in upd:
                msg = upd["message"]
                if msg.get("text") == "/start":
                    bot.send_message(
                        msg["chat"]["id"],
                        "👋 یک گزینه:",
                        reply_markup={"inline_keyboard": [
                            [{"text": "🕐 ساعت", "callback_data": "time"}],
                            [{"text": "🎲 تاس", "callback_data": "dice"}],
                        ]}
                    )
            elif "callback_query" in upd:
                cq = upd["callback_query"]
                if cq.get("data") == "time":
                    bot.answer_callback_query(cq["id"],
                        text=f"🕐 {datetime.datetime.now():%H:%M}")
                elif cq.get("data") == "dice":
                    bot.answer_callback_query(cq["id"])
                    bot.send_dice(chat_id=cq["message"]["chat"]["id"], emoji="🎲")

if __name__ == "__main__":
    main()
04 — ربات خوش‌آمد + فیلتر لینکمدیریت
به اعضای جدید خوش‌آمد می‌گه و لینک رو حذف می‌کنه.
import re
from gitbin import Bot

TOKEN = "YOUR_BOT_TOKEN"
bot = Bot(token=TOKEN)
LINK_RE = re.compile(r"https?://|t\.me/|www\.")

def main():
    bot.delete_webhook()
    offset = 0
    while True:
        updates = bot.get_updates(offset=offset, timeout=25)
        for upd in updates or []:
            offset = upd["update_id"] + 1
            if "message" not in upd:
                continue
            msg = upd["message"]
            chat_id = msg["chat"]["id"]
            text = msg.get("text", "")
            message_id = msg.get("message_id")

            for m in msg.get("new_chat_members") or []:
                bot.send_message(chat_id,
                    f"👋 خوش اومدی {m.get('first_name','کاربر')}!")

            if text and LINK_RE.search(text):
                bot.delete_message(chat_id, message_id)
                bot.send_message(chat_id, "🚫 لینک مجاز نیست.")

if __name__ == "__main__":
    main()

✅ هر description یک Exception اختصاصی داره.

try/exceptمثال
from gitbin import Bot, GitbinError
from gitbin.errors import ChatIdInvalid, BalanceTooLow

try:
    bot.send_message(chat_id="bad", text="x")
except ChatIdInvalid as e:
    print("چت نامعتبر:", e)
